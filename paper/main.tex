\section{Empirical Strategy}
\label{sec:strategy}
%==========================================================

% TODO: Suggested reviewer framing for this section:
% "Our baseline design exploits within-municipality changes in local income tax
% rates in a municipality-by-year panel with municipality and year fixed effects.
% Because tax changes may coincide with broader local shocks, we use two
% complementary strategies. First, among municipalities that change their tax rates,
% we estimate a dose-response specification that separates common post-change effects
% from the marginal effect of the rate increase itself. Second, we implement
% event-study and stacked difference-in-differences designs using nearby municipalities
% with stable tax rates as controls."

\subsection{Baseline Specification}
\label{subsec:baseline}

\begin{equation}
    \text{share\_live\_in}_{it} = \alpha_i + \gamma_t + \beta \cdot \text{rate}_{it} + \varepsilon_{it}
    \label{eq:twfe}
\end{equation}

\noindent where:
\begin{itemize}
    \item[] $\text{share\_live\_in}_{it}$ = fraction of workers in municipality $i$ at time $t$ who also reside there
    \item[] $\alpha_i$, $\gamma_t$ = municipality and year fixed effects
    \item[] $\text{rate}_{it}$ = municipal income tax rate
    \item[] Standard errors clustered at the municipality level
\end{itemize}

\subsection{Dose-Response Identification}
\label{subsec:dose}

\begin{equation}
    \text{share\_live\_in}_{it} = \alpha_i + \gamma_t + \delta \cdot \text{post}_{it} + \beta \cdot (\text{rate}_{it} \times \text{post}_{it}) + \varepsilon_{it}
    \label{eq:dose}
\end{equation}

\noindent where:
\begin{itemize}
    \item[] $\text{post}_{it}$ = indicator for post-first-increase period
    \item[] $\delta$ captures common post-treatment effects among rate-changing municipalities
    \item[] $\beta$ captures the tax-specific channel: conditional on raising taxes, does rate magnitude predict additional sorting?
\end{itemize}
