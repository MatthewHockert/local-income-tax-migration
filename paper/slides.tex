\section{Introduction}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}
\fontfamily{cmr}
\frametitle{Motivation}
    \begin{itemize}\setlength{\itemsep}{0.25cm}
        \item Tiebout (1956): citizens sort across jurisdictions
        according to their demands for local public goods.
        \item Strong evidence of tax-driven sorting at the state level
        \citep{kleven2020, young2016}.
        \item Little causal evidence on local income tax sorting within
        metro areas
        \item Two data problems make this hard. Local tax jurisdictions
        are sub-county, so IRS/ACS migration data are too coarse.
        Credit systems and weak enforcement complicate the effective
        tax burden.
    \end{itemize}
\end{frame}

\begin{frame}
\fontfamily{cmr}
\frametitle{This Paper}
    \begin{itemize}\setlength{\itemsep}{0.2cm}
        \item Question: Do local income tax rate changes cause sorting?
        \item Setting: Pennsylvania, 2002--2019. Universal Earned
        Income Tax with continuous rate variation across $\sim$2{,}500
        munis $\times$ $\sim$500 school districts.
        \item Data: LEHD LODES origin--destination, linking
        residence tract with workplace tract annually.
        \item Outcomes: \texttt{share\_live\_in}
        = (workers who live and work in city $i$) / (total workers in $i$),
        plus count outcomes (resident workers, total jobs).
        \item Exogeneous identification: Act 32
        consolidated collection and made employer withholding
        mandatory.
    \end{itemize}
\end{frame}

\begin{frame}
\fontfamily{cmr}
\frametitle{Institutional Setting (Pennsylvania)}
    \begin{itemize}\setlength{\itemsep}{0.25cm}
        \item Universal Earned Income Tax. About 2{,}500 munis and 500
        school districts each set their own rate. Typical 0.5\% muni
        + 0.5\% school district.
        \item Before Act 32, collection was fragmented and withholding
        was inconsistent.
        \item Act 32 (effective 2012) consolidated collection and
        required employers to withhold the higher of the home and work
        rates.
    \end{itemize}
\end{frame}

%%%%%%%%%%% How PA tax works -- piecewise %%%%%%%%%%%
\begin{frame}
\fontfamily{cmr}
\frametitle{How the PA EIT works (post-Act 32, excluding Philadelphia)}
    \small
    A worker pays the higher of the total home rate (muni resident plus
    school district) and the nonresident rate at the work location.
    The bill is split piecewise:
    \[
        \bigl(\tau^{\text{home}},\ \tau^{\text{work}}\bigr) =
        \begin{cases}
            \bigl(\tau^{\text{res}}_j + \tau^{\text{SD}}_j - \tau^{\text{nr}}_i,\ \tau^{\text{nr}}_i\bigr)
              & \text{if } \tau^{\text{res}}_j + \tau^{\text{SD}}_j > \tau^{\text{nr}}_i \\[2pt]
            \bigl(0,\ \tau^{\text{nr}}_i\bigr)
              & \text{if } \tau^{\text{res}}_j + \tau^{\text{SD}}_j \leq \tau^{\text{nr}}_i
        \end{cases}
    \]
    \begin{itemize}\setlength{\itemsep}{0.1cm}
        \item The work jurisdiction always collects its full nonresident rate.
        \item The home muni and school district then split whatever remains
        of the worker's bill. They get nothing if the work rate is higher.
        \item Which rate binds determines how the worker can respond.
        When the home rate binds, the worker can lower the bill by moving
        home. When the work rate binds, the worker can lower it only by
        changing jobs.
    \end{itemize}
\end{frame}

%%%%%%%%%%% Worked numeric example %%%%%%%%%%%
\begin{frame}
\fontfamily{cmr}
\frametitle{Worked example}
    \small
    Worker 1 lives in city $B$ and works in city $A$.
    Home muni rate is 1.0\%, home school district is 0.5\%, so the total
    home rate is 1.5\%. The work-location nonresident rate is 0.8\%.
    \begin{itemize}\setlength{\itemsep}{0.1cm}
        \item Worker 1 pays the higher of the two rates, so 1.5\%.
        Home collects $1.5 - 0.8 = 0.7$\%; work collects 0.8\%.
        \item This is a home-binds case. To lower the bill,
        worker 1 would have to move out of $B$. Changing jobs would
        not help because the work rate is already lower.
    \end{itemize}
\end{frame}

%%%%%%%%%%% Philadelphia handling %%%%%%%%%%%
\begin{frame}
\fontfamily{cmr}
\frametitle{Philadelphia: a separate regime}
    \small
    \begin{itemize}\setlength{\itemsep}{0.15cm}
        \item Philadelphia is exempt from Act 32. Its Wage Tax operates
        under the Sterling Act of 1932, administered by the City
        Department of Revenue.
        \item Philly residents pay the Philly resident rate regardless
        of where they work. There is no max rule and no offset.
        \item Nonresidents working in Philly pay the Philly nonresident
        rate, which goes entirely to Philly.
        \item In my analysis I keep Philly in the main sample because
        it is part of PA's labor market, but exclude it in robustness
        checks because its reporting and dynamics differ from Act-32 munis.
    \end{itemize}
\end{frame}

\begin{frame}
\fontfamily{cmr}
\frametitle{Three Channels of Adjustment}
    When muni $A$ raises its rate, residents of $A$ can:
    \vspace{0.15cm}
    \begin{enumerate}\setlength{\itemsep}{0.18cm}
        \item Move residence to a lower-rate muni, keep job in $A$.
        Saves at least the home SD rate at $A$.
        \item Move both residence and job to a lower-rate muni.
        Largest savings.
        \item New entrants pick a home outside $A$ for a given job.
    \end{enumerate}
    \vspace{0.25cm}
    All three reduce \texttt{share\_live\_in} for $A$.
\end{frame}

%%%%%%%%%%% Q2 (intensive margin -- the headline) %%%%%%%%%%%
\begin{frame}
\fontfamily{cmr}
\frametitle{Q2: Do rate increases cause sorting?}
    Two specs
    \[
    \text{Total stack:}\quad Y_{it} = \alpha_i + \gamma_t + \beta\,\tau^{\text{home}}_{it} + \mathbf{X}'_{it}\boldsymbol{\phi} + \varepsilon_{it}
    \]
    \[
    \text{Decomposition:}\quad Y_{it} = \alpha_i + \gamma_t + \beta^{m}\tau^{\text{res,muni}}_{it} + \beta^{sd}\tau^{\text{SD}}_{it} + \mathbf{X}'_{it}\boldsymbol{\phi} + \varepsilon_{it}

    \footnotesize
    \begin{itemize}
    \item Treated = munis that ever raised their muni or SD rate (within-city
    rate change drives identification). 
    \item Control = munis whose rates
    stayed flat over the panel. 
    \item $\mathbf{X}_{it}$ = log muni spending
    on police, streets, debt, state aid. $Y_{it}$ = \texttt{share\_live\_in}
    or log count outcomes.
    \end{itemize}
    
\end{frame}

\begin{frame}
\fontfamily{cmr}
\frametitle{Mechanism: Is it enforcement? (Act 32)}
    \small
    \textbf{Unit $i$ = PA municipality; $t$ = year.} Outcome
    \texttt{share\_live\_in}$_{it}$. Structural break on rate effect:
    \[
    Y_{it} = \alpha_i + \gamma_t
    + \beta^{\text{pre}} \tau_{it} \cdot \mathbf{1}[t<2012]
    + \beta^{\text{post}} \tau_{it} \cdot \mathbf{1}[t\geq 2012]

    \]

        \item Pre-2012: fragmented collection, weak withholding;

        of munis reported nonres rates; jumped to $\sim$72\% in 2012).
        \item Post-2012: centralized, mandatory employer
        withholding at $\max(\tau^{\text{home}}, \tau^{\text{nr,work}})$.
        \item Run break for both rates separately to see which channel
        is enforcement-bound.
    \end{itemize}
\end{frame}

\begin{frame}
\fontfamily{cmr}
\frametitle{Do the rate increases we exploit actually bind?}
    \small
    \textbf{For self-pair (worker lives \emph{and} works in muni $X$):}
    home stack $= \tau^{\text{res}}_X + \tau^{\text{SD}}_X$ vs.\ work
    nonres rate $\tau^{\text{nr}}_X$.
    \begin{itemize}\setlength{\itemsep}{0.10cm}
        \item Across all PA muni-years post-Act-32 (15{,}423):
        home binds 41\%, work binds 59\%. Mean wedge
        $\tau^{\text{home}} - \tau^{\text{nr}} = +0.27$ pp.
        \item Of muni-resident rate increases 2012--2019 (57 events):
        \textbf{84\% (48 of 57)} occur where home binds at the muni
        $\Rightarrow$ the rate increase \emph{moves the bill} for
        residents working locally.
        \item The remaining 16\% (9 of 57) occur where work already
        binds and stays binding $\Rightarrow$ the rate hike is
        irrelevant to in-muni workers' bills.
    \end{itemize}
    \vspace{0.1cm}
    \textbf{Cross-pair within 80 km, post-Act-32 (1.83M pair-years):}
    \begin{itemize}\setlength{\itemsep}{0.04cm}
        \item Work binds (home stack $\le$ work nonres): 61.6\%.
        \item Home binds (home stack $>$ work nonres): 38.4\%.
    \end{itemize}
    \vspace{0.1cm}
    \footnotesize \emph{Implication}: most of the resident-rate
    increases we exploit are economically meaningful for the
    in-muni worker population. The piecewise binding test below
    sharpens this: separates extensive switches in binding side from
    intensive within-channel rate changes.
\end{frame}

        \item Control: never-changed munis (1{,}110).
        \item Flip-floppers dropped.
        \item Continuous rate (per pp), not binary cohort.
        \item $\mathbf{X}_{it}$ = log muni spending (police, streets,
        debt, state aid) + revenue shares (EIT, real estate, intergov).
        \item $\tau^{\text{res,muni}}:$ nc dropped (no within-variation by definition).
    \end{itemize}
\end{frame}

\begin{frame}
\fontfamily{cmr}
\frametitle{Q3 Robustness motivation: staggered + multi-event treatment}
    \small
    \begin{itemize}\setlength{\itemsep}{0.10cm}
        \item Of 113 inc-only munis, 17 ($\approx$15\%) raise rates more

%%%%%%%% (2) DCDH %%%%%%%%
\begin{frame}

%%%%%%%% PIECEWISE BINDING MODEL %%%%%%%%
\begin{frame}
\fontfamily{cmr}
\frametitle{(5) Piecewise binding model: setup}
    \scriptsize
    Cousub-pair panel (work $A$, home $B$, year $t$). Under the
    institutional rule, the work jurisdiction always collects
    $\tau^{\text{nr}}_A$; the home jurisdictions collect the
    \emph{residual} $\max(\tau^{\text{home}}_B - \tau^{\text{nr}}_A,\,0)$.
    Worker's total bill:
    \[
        \tau^{\text{bill}}(A,B,t)
        \;=\; \max\!\bigl(\tau^{\text{home}}_B,\; \tau^{\text{nr}}_A\bigr),
        \qquad
        \tau^{\text{home}}_B = \tau^{\text{res}}_B + \tau^{\text{SD}}_B.
    \]
    \emph{Binding side} = which jurisdiction's rate moves the bill at the
    margin: \textbf{home binds} iff $\tau^{\text{home}}_B > \tau^{\text{nr}}_A$,
    else \textbf{work binds}.
    For each pair and each year-over-year transition $(t-1)\!\to\!t$
    (pooled across all $t \in 2013\text{--}2019$), classify:
    \begin{itemize}\setlength{\itemsep}{0.04cm}
        \item \textbf{int\_home}: home-binds in both $t-1$ and $t$.
        \item \textbf{int\_work}: work-binds in both $t-1$ and $t$.
        \item \textbf{ext\_to\_home}: was work-binds, now home-binds.
        \item \textbf{ext\_to\_work}: was home-binds, now work-binds.
    \end{itemize}
    Decomposition of the year-over-year change in the bill:
    \begin{align*}
        \Delta\log\text{flow}_{ABt} &=
        \beta_{ih}\,\Delta\tau^{\text{bill}}_{ABt}\!\cdot\!\mathbf{1}\{\text{int\_home}\}
      + \beta_{iw}\,\Delta\tau^{\text{bill}}_{ABt}\!\cdot\!\mathbf{1}\{\text{int\_work}\} \\
      &\quad
      + \beta_{eh}\,\Delta\tau^{\text{bill}}_{ABt}\!\cdot\!\mathbf{1}\{\text{ext\_to\_home}\}
      + \beta_{ew}\,\Delta\tau^{\text{bill}}_{ABt}\!\cdot\!\mathbf{1}\{\text{ext\_to\_work}\} \\
      &\quad + \alpha_{AB} + \gamma_{At} + \varepsilon_{ABt}
    \end{align*}
    Each $\beta$ is per-pp dose response within that channel-state.
    Pair FE $\alpha_{AB}$ + work-muni $\times$ year FE $\gamma_{At}$
    absorbs muni-$A$-specific year shocks (e.g., booming labor
    market that drives flows up while $A$ raises rates). SE clustered
    at work muni.
\end{frame}

\frametitle{Putting it Together (1/2): Who moves?}
    \small
    \textsc{Setup.} A post-Act 32 PA city: 1{,}000 total workers;
    150 live-and-work there (share\_live\_in = 15\%); EIT rises
    0.5\% $\rightarrow$ 1.5\%.
    \vspace{0.2cm}
    \begin{itemize}\setlength{\itemsep}{0.15cm}
        \item[Q2/Q3.] Over 7--8 years, share falls to $\sim$13.2\% $\to$
        $\sim$18 of 150 locally-resident workers gone. If the city later
        cuts the rate back, none of them return.
        \item[Q4.] Those 18 leave the city entirely (not a neighborhood move);
        nearby lower-rate munis within 80 km absorb them.
        \item[Q5.] If the overlapping school district had raised its rate
        instead, residents would have moved in, not out.
        \item[Q6.] The loss is spread across the city, not concentrated
        at municipal boundaries --- no ``edge effect.''
    \end{itemize}

\begin{frame}

    \small
    Same city: 1{,}000 workers, rate 0.5\% $\rightarrow$ 1.5\%, $\sim$18 residents lost.
    \vspace{0.2cm}
    \begin{itemize}\setlength{\itemsep}{0.15cm}
        \item[Fiscal.] EIT revenue roughly doubles ($+67$ log points);
        total tax $+30\%$. Property values and population: unchanged.
        \item[Q7.] Most new dollars flow to community development
        (planning, econ dev, housing) --- not to police patrols residents see.
        \item[Enforcement.] None of this happens pre-Act 32: same statutory
        rate, precise zero behavioral response. Collection capacity is the
        binding constraint.
    \end{itemize}

    doubling of EIT revenue. Rate increases work --- when enforced.

\begin{frame}
\fontfamily{cmr}
\frametitle{Summary of Findings}
    \begin{enumerate}\setlength{\itemsep}{0.15cm}
        \item Higher muni rates reduce \texttt{share\_live\_in} by about
        1.3 share-points per 1 pp, building over 7 to 8 years.
        \item The effect is enforcement-contingent. The muni rate
        coefficient is a precise zero before Act 32 and roughly
        $-1.8$ share-points after.
        \item The response is asymmetric. Increases push residents out,
        but decreases do not bring them back.
        \item Muni and school district rates have opposite signs.
        Bundling them as a single "total EIT" rate hides both channels.
    \end{enumerate}
\end{frame}
