%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Introduction}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

    %
    \begin{frame}
    \fontfamily{cmr}
    \frametitle{Question}
        \begin{itemize}\setlength{\itemsep}{0.15cm}
            \item When a Pennsylvania municipality changes its
            earned-income-tax rate, do workers \emph{sort} ---
            relocating residence or workplace?
            \item Three margins of response, all observable in LODES
            origin-destination data:
            \begin{itemize}\setlength{\itemsep}{0.05cm}
                \item \textbf{Live-and-work} workers: residents employed locally.
                \item \textbf{In-commuters}: nonresidents employed in the muni.
                \item \textbf{Out-commuters}: residents employed elsewhere.
            \end{itemize}
            \item Empirical challenge: the bill is piecewise
            $T = \max(\tau^{\text{home}},\, \tau^{\text{nr,work}})$.
            A rate change only \emph{moves the bill} when its side binds.
        \end{itemize}
    \end{frame}
    %

    %
    \begin{frame}
    \fontfamily{cmr}
    \frametitle{Roadmap}
        \begin{itemize}\setlength{\itemsep}{0.15cm}
            \item \textbf{Question:} Do PA EIT rate changes cause residential / job sorting?
            \item \textbf{Data:} LODES OD (2002--2019, PA) + DCED EIT panel + AFR fiscal panel.
            \item \textbf{Methods:} One muni $\times$ year panel; both
            $\Delta\tau^{\text{res}}$ and $\Delta\tau^{\text{nr}}$ in the same regression;
            decomposed by direction $\times$ binding state. Five estimators.
            \item \textbf{This deck:} Setup $\to$ cell counts $\to$ headline
            results $\to$ triangulation $\to$ event studies $\to$ placebo + robustness.
        \end{itemize}
    \end{frame}
    %

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Background}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

    %
    \begin{frame}
    \fontfamily{cmr}
    \frametitle{Institutional Setting (PA EIT, post-Act 32)}
        \footnotesize
        For a PA resident working outside Philadelphia:
        \[
            T_{ij} = \max\!\bigl(\tau^{\text{res}}_j + \tau^{\text{SD}}_j,\; \tau^{\text{nr}}_i\bigr).
        \]
        Liability allocation: work muni collects up to its nonres rate;
        home muni + SD collect any residual.
        \medskip

        \begin{itemize}\setlength{\itemsep}{0.10cm}
            \item \textbf{Act 32 (2008 / effective 2012)} standardized
            employer withholding, residency certification, and consolidated
            tax-collection districts. Statutory rates were unchanged ---
            the change was administrative.
            \item \textbf{Empirical implication.} Pre-Act-32, the same
            statutory rate was weakly enforced. Post-Act-32, the binding
            rule above actually moves the bill.
            \item \textbf{Sample.} 2002--2019 (avoid COVID).
        \end{itemize}
    \end{frame}
    %

    %
    \begin{frame}
    \fontfamily{cmr}
    \frametitle{Data}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{0.10cm}
            \item \textbf{LODES OD} (2002--2019, PA): block-level
            origin-destination flows aggregated to PA county subdivisions
            (cousubs) via TIGER 2020 block centroid. Recovers all
            2{,}561 PA cousubs (vs 1{,}479 under the prior tract-centroid join).
            \item \textbf{DCED EIT panel}: muni $\times$ year resident,
            school district, and nonresident rates. Reparsed XLS to
            recover SD identifiers. Nonres rate has reporting gaps
            (handled via three sample variants).
            \item \textbf{DCED AFR}: muni fiscal panel for controls
            (share-based: police / streets / debt service as a share of
            total expenditures; EIT / real estate / intergov as a share
            of total revenues).
            \item \textbf{Sample}: $\sim$2{,}500 cousubs $\times$ 18 years.
        \end{itemize}
    \end{frame}

\section{Methods}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

    %
    \begin{frame}
    \fontfamily{cmr}
    \frametitle{Setup: One Panel, Three Treatments}
        \footnotesize
        \textbf{Unit} $i$: a PA cousub. \textbf{Time} $t$: year.
        One row per $(i,t)$.
        \medskip

        \textbf{Outcomes per row} (eight; covered simultaneously):
        \begin{itemize}\setlength{\itemsep}{0.02cm}\scriptsize
            \item \texttt{log\_live\_and\_work}, \texttt{log\_in\_commuters},
            \texttt{log\_out\_commuters}
            \item \texttt{log\_total\_jobs}, \texttt{log\_total\_residents}
            \item \texttt{share\_live\_in}, \texttt{share\_nonres},
            \texttt{share\_work\_outside}
        \end{itemize}
        \medskip

        \textbf{Three treatments per row}, each separately decomposed by cells:
        \begin{itemize}\setlength{\itemsep}{0.02cm}\scriptsize
            \item $\Delta\tau^{\text{muni}}_{it}$: cumulative muni resident-rate change.
            \item $\Delta\tau^{\text{SD}}_{it}$: cumulative school-district rate change.
            \item $\Delta\tau^{\text{nr}}_{it}$: cumulative muni nonresident-rate change.
        \end{itemize}
        \medskip

        \textbf{Per-row binding state}:
        $\text{home\_binds}_{it} = \mathbf{1}\{\tau^{\text{home}}_{it} > \tau^{\text{nr}}_{it}\}$
        where $\tau^{\text{home}} = \tau^{\text{muni}} + \tau^{\text{SD}}$.
        Cells classified separately for each treatment using its own first event.
    \end{frame}
    %

    %
    \begin{frame}
    \fontfamily{cmr}
    \frametitle{Cell Classification (per side, fixed at first event)}
        \footnotesize
        Each muni gets one cell label on each side, based on the
        binding state \emph{just before} vs \emph{just after} its first event.
        \medskip
        \scriptsize
        \renewcommand{\arraystretch}{1.20}
        \setlength{\tabcolsep}{6pt}
        \begin{tabular}{lll}
            \toprule
            Cell & Direction & Binding transition \\
            \midrule
            \textcolor{GopherMaroon}{\textbf{Inc-bind}}     & Up   & binding pre and post (bill $\uparrow$ by full $\Delta\tau$) \\
            \textcolor{GopherMaroon!70}{\textbf{Inc-becomes}} & Up   & flips non-binding $\to$ binding (bill rises by $\tau_{\text{post}} - \tau^{\text{nr}}$) \\
            \textcolor{DarkGray}{\textbf{Inc-nobind}}        & Up   & non-binding throughout (\textbf{bill unchanged} --- placebo) \\
            \textcolor{GopherDarkMaroon}{\textbf{Dec-bind}}  & Down & binding pre and post (bill $\downarrow$ by full $\Delta\tau$) \\
            \textbf{Dec-becomes-non}                          & Down & flips binding $\to$ non-binding \\
            \textcolor{DarkGray}{\textbf{Dec-nobind}}         & Down & non-binding throughout (\textbf{bill unchanged} --- placebo) \\
            NC                                                & ---  & never-changed (control) \\
            \bottomrule
        \end{tabular}
        \medskip
        \footnotesize
        Institutional rule predicts |Inc-bind| $>$ |Inc-becomes| $>$ |Inc-nobind|
        (= 0). Inc-nobind and Dec-nobind are \textbf{predicted-null
        placebos within the design.}
    \end{frame}
    %

    %
    \begin{frame}
    \fontfamily{cmr}
    \frametitle{Headline Estimator: Stacked Continuous DiD}
        \scriptsize
        \alert{Headline equation} (run on the stacked panel; one row per
        muni-stack-year, NC munis appear in every stack as clean controls):
        \begin{equation*}
        \begin{split}
        Y_{\tilde{i}\tilde{t}} \;=\; \alpha_{\tilde{i}} \;+\; \gamma_{\tilde{t}}
          &+ \sum_{k\in\mathcal{C}\setminus\text{NC}}\beta^{\text{muni}}_k\,\Delta\tau^{\text{muni}}_{it}\cdot\mathbf{1}\{\text{cell}^{\text{muni}}_i=k\} \\
          &+ \sum_{k\in\mathcal{C}\setminus\text{NC}}\beta^{\text{SD}}_k\,\Delta\tau^{\text{SD}}_{it}\cdot\mathbf{1}\{\text{cell}^{\text{SD}}_i=k\} \\
          &+ \sum_{k\in\mathcal{C}\setminus\text{NC}}\beta^{\text{nr}}_k\,\Delta\tau^{\text{nr}}_{it}\cdot\mathbf{1}\{\text{cell}^{\text{nr}}_i=k\}
          + \mathbf{X}'_{it}\phi + \varepsilon
        \end{split}
        \end{equation*}
        \scriptsize
        \begin{itemize}\setlength{\itemsep}{0.02cm}
            \item $\tilde{i}=(\text{stack},\,\text{muni})$, $\tilde{t}=(\text{stack},\,\text{year})$
            so each muni has a separate intercept per stack.
            \item $\mathcal{C}\setminus$NC = \{Inc-bind, Inc-becomes, Inc-nobind,
            Dec-bind, Dec-becomes-non, Dec-nobind\}; NC is omitted reference \emph{per side}.
            \item Each muni $i$ appears in exactly one sub-experiment (its own cohort)
            on each side where it's non-NC; NC munis appear in every cohort.
            \item Each $\beta^s_k$ is the per-pp dose-response of $Y$ to a 1 pp change
            in rate $s$, restricted to munis whose first event on side $s$ is in cell $k$,
            \emph{relative to NC's contemporaneous trend on side $s$}.
            \item Within each sub-experiment, comparison is treated cohort vs NC
            in the same calendar-year window. Treated munis from \emph{other}
            cohorts are not used as controls --- avoids Goodman-Bacon negative weighting.
            \item Stack-muni and stack-year FE absorb cohort-specific level/time effects.
        \end{itemize}
    \end{frame}
    %

    %
    \begin{frame}
    \fontfamily{cmr}
    \frametitle{What's Being Compared}
        \footnotesize
        For each sub-experiment $\mathcal{S}^s_c$ (one per cohort year $c$ on side $s$):
        \begin{itemize}\setlength{\itemsep}{0.05cm}\scriptsize
            \item \textbf{Treated set:} munis $i$ with $c^s_i = c$ and cell$^s_i \neq$ NC.
            \item \textbf{Control set:} \emph{all} NC$^s$ munis (truly stable rate on side $s$).
            \item \textbf{Window:} $t \in [c-5, c+5]$.
            \item \textbf{Dose:} $\Delta\tau^s_{it} = \tau^s_{it} - \tau^s_{i,c-1}$ for treated $i$;
            $0$ for NC munis at all $t$ and for treated $i$ at $t < c$.
        \end{itemize}
        \medskip

        \textbf{Within-design placebo.} The Inc-nobind cell --- rate hike that
        leaves the bill unchanged because home stack stays below nonres rate ---
        is the institutional rule's predicted-null cell. If it shows zero behavioral
        response while Inc-bind shows large response, the response is mechanism-driven.
        \medskip

        \textbf{Five estimators.} Q2 TWFE pooled $\to$ single-event TWFE ($\le$2 events) $\to$
        stacked binary (cell $\times$ post) $\to$ \alert{\textbf{stacked continuous}
        (cell $\times \Delta\tau$, \alert{headline})} $\to$ DCDH (per cell).
        \medskip

        \textbf{Controls.} Share-based, all time-varying:
        \texttt{share\_police}, \texttt{share\_streets}, \texttt{share\_debt\_svc}
        (denom = total expenditures); \texttt{share\_eit}, \texttt{share\_re},
        \texttt{share\_intergov} (denom = total revenues).
        SE clustered at muni level.
    \end{frame}
    %

    %
    \begin{frame}
    \fontfamily{cmr}
    \frametitle{Three Sample Variants for Missing Nonres Rates}
        \footnotesize
        DCED's nonres-rate reporting is patchy pre-Act-32 ($\sim$55\%
        coverage) and incomplete post ($\sim$72--80\%). All else equal,
        we run three samples:
        \medskip
        \scriptsize
        \renewcommand{\arraystretch}{1.20}
        \setlength{\tabcolsep}{8pt}
        \begin{tabular}{ll}
            \toprule
            Sample & Treatment of NA \texttt{muni\_nonres\_eit} \\
            \midrule
            \textbf{strict} & Drop muni-years with NA nonres rate. \\
            \textbf{filled} & Impute \texttt{nonres = resident\_eit} when NA (statute-anchored). \\
            \textbf{drop}   & Keep all years; drop NA at regression time per outcome. \\
            \bottomrule
        \end{tabular}
        \medskip
        \footnotesize
        \textbf{Headline tables}: \texttt{drop} sample (largest N).
        \texttt{filled} and \texttt{strict} reported as robustness.
    \end{frame}

    \begin{frame}
    \fontfamily{cmr}
    \frametitle{Reading the Spillover Pattern}
        \footnotesize
        \textbf{What we see:} ring-1 neighbors of Inc-bind munis \emph{lose}
        live-and-work workers ($-15\%$); ring-2 neighbors \emph{gain} ($+9\%$).
        Pattern is robust to lower-rate restriction.
        \medskip

        \textbf{What this rules out.} The "fiscal-advantage leapfrog" reading
        (leavers specifically target lower-rate ring-2 munis) should produce
        bigger coefficients under the lower-rate restriction. It doesn't.
        Spillover pattern is geographic, not fiscal-differential-driven.
        \medskip

        \textbf{Most consistent reading: \emph{spatial labor-market clustering}.}
        Ring-1 munis share X's labor market and consumer-services demand. When X
        enters Inc-bind:
        \begin{itemize}\setlength{\itemsep}{0.02cm}\scriptsize
            \item Consumer-facing jobs in X fall 29\%. Ring-1 residents who
            worked in X (or in similar nearby munis) lose those jobs $\to$
            drop out of \texttt{live\_and\_work} in their muni.
            \item Ring-1 munis share fiscal-pressure shocks with X (clustered geography).
            \item Ring-2 is far enough that labor-market spillover is muted, but
            close enough to absorb residential moves --- $+9\%$ live-and-work
            consistent with new residents who got new ring-2 jobs.
        \end{itemize}
        \medskip

        \textbf{Open extension (highway-network refinement).} Are ring-2 gains
        concentrated along major highway corridors connecting back to X? If
        leavers retain their X jobs and commute back, gains should be highest
        in ring-2 munis on a highway corridor. Requires TIGER primary roads pull.
    \end{frame}

    \begin{frame}
    \fontfamily{cmr}
    \frametitle{Reading the Highway Test}
        \footnotesize
        \textbf{Result is the opposite of the highway-leapfrog prediction.}
        Ring-2 absorption ($+0.100^{**}$) is concentrated in
        \emph{non-corridor-shared} neighbors. Cousubs that share an
        interstate corridor with treated A do \emph{not} absorb leavers.
        \medskip

        \textbf{Interpretation: highway corridors connect labor markets.}
        A ring-2 muni sharing an interstate with A is in the same regional
        economy. When A's consumer-facing jobs fall 29\%, corridor-shared
        ring-2 munis face the same demand shock --- they're losing
        residents too, not absorbing A's leavers.
        \medskip

        \textbf{Refined story.} A's rate hike pushes residents out. They
        land in ring-2 munis that are nearby \emph{but economically separable}
        (no shared interstate corridor with A). They get \emph{new local jobs}
        in those ring-2 munis ($+0.100^{**}$ on \texttt{live\_and\_work},
        not \texttt{out\_commuters}). \textbf{Not "move and commute back."}
        \medskip

        \textbf{Caveat.} Corridor pairs are static (TIGER 2015 = 2020).
        Power is limited (24--26 corridor-receiver-years per year). Read
        the magnitude direction; precision is bounded.

    \begin{frame}
    \fontfamily{cmr}
    \frametitle{On Stacked DiD and Treatment Overlap}
        \footnotesize
        \textbf{Common concern:} ``Stacked DiD with continuous treatment
        cannot have overlapping treatment times.''
        \medskip

        \textbf{Reading.} The constraint is \emph{within each sub-experiment,
        controls must be clean.} Calendar-year overlap across cohorts is fine
        (and unavoidable in any staggered design).
        \medskip

        \begin{itemize}\setlength{\itemsep}{0.05cm}\scriptsize
            \item Each sub-experiment $cy$ uses treated munis from cohort
            $cy$ + NC munis only. Treated munis from \emph{other} cohorts
            are not included as controls.
            \item Within-sub-experiment FE (\texttt{stack\_muni},
            \texttt{stack\_year}) absorb stack-specific level/time effects.
            \item Each muni appears in exactly one stack as treated (its own
            cohort), or in every stack as NC if never-treated.
            \item No muni-year is double-counted as both treated and control.
        \end{itemize}

        \textbf{What goes wrong with vanilla TWFE under staggered treatment:}
        later-treated units used as controls for earlier-treated cohorts
        (Goodman-Bacon 2021). Stacked DiD avoids this by construction.
        \textbf{Our design is correct.}
    \end{frame}

            \textbf{Inc-bind, FULL ($n$=40)} & $+0.060$  & $+0.008$  & $+0.057$  & $-0.003$  & \alert{Null --- clean} \\
            \textbf{Inc-bind, CITY only ($n$=8)} & $\bm{+0.194^{***}}$ & $\bm{+0.105^{**}}$ & $\bm{+0.118^{**}}$ & $\bm{+0.101^{***}}$ & \alert{Present} \\

        \textbf{The pooled Inc-bind headline} ($-0.402$): pre-trends are null
        across $t=-5$ to $t=-2$. Parallel trends roughly hold. Headline is
        defensible.
        \medskip

        \textbf{City-specific finding} ($-0.455$): pre-trends \emph{are}
        significant. The 8 cities had levels above the NC city pool in the
        pre-period, declining toward NC by $t=-1$. City-specific magnitude
        claim must be hedged.
        \medskip

        \textbf{Placebo cell} (Inc-nobind, $-0.062$): small but significant
        pre-trends. Imperfect placebo. Worth disclosing as a caveat.
    \end{frame}

    \begin{frame}
    \fontfamily{cmr}
    \frametitle{Honest Causal Read}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{0.10cm}
            \item \textbf{Pooled Inc-bind cell (40 munis): defensible.} Null
            pre-trends in binary SA. Null per-pp dose pre-trends by
            construction. Post-event coefficients (\texttt{stacked\_continuous}
            $-0.402^{***}$) plausibly causal under standard parallel-trends.
            \item \textbf{City-specific magnitude (8 munis): \alert{do not claim
            as causal}.} Pre-trends are present; post-event drop is partly
            continuation of differential trajectory. Frame as ``rate hikes in
            declining cities \emph{associated with} continued decline.''
            \item \textbf{Placebo (Inc-nobind, 26 munis):} small pre-trends mean
            the within-design placebo isn't perfectly clean. Still null on
            \texttt{log\_live\_and\_work} post-event ($-0.062$, insig).
            Acknowledge as caveat.
            \item \textbf{Sharpening for the city result (open follow-ups).}
            (i) Matched controls --- pair each Inc-bind city with NC cities
            on similar pre-2002 trajectories.
            (ii) Synthetic control per Inc-bind city.
            (iii) Drop steepest-pre-trend cities (Wilkes-Barre, Hazleton)
            and re-estimate; check robustness to specific inclusion.
        \end{itemize}
    \end{frame}

    \begin{frame}
    \fontfamily{cmr}
    \frametitle{Reading the Industry Decomposition}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{0.10cm}
            \item \textbf{Consumer-facing jobs fall 29\%} when the muni resident rate
            hikes (Inc-bind cell). Retail, accommodation/food, arts, other services
            --- all sectors whose demand depends on local population.
            \item \textbf{Tradeable jobs are insulated.} Manufacturing, wholesale,
            information, finance, professional services. Point estimate is
            actually \emph{positive} ($+0.244$, insig). These sectors sell to
            regional/national markets and don't depend on local residents.
            \item \textbf{Composition shifts toward tradeables}: share\_tradeable
            $+11$pp on Inc-bind. The consumer-facing denominator shrinks faster
            than the tradeable numerator.
            \item \textbf{The synthesis becomes positive, not just ``consistent with.''}
            Total-jobs decline operates \emph{via} local-services demand spillover
            from residents leaving --- direct evidence, not ruled-out alternatives.
            \item \textbf{Placebo wrinkle}: Inc-nobind tradeable is $-0.273^{**}$
            (consumer-facing $-0.028$, null). Could be small-$n$ (26 munis) or
            signaling. The demand-spillover story holds for consumer-facing
            specifically; tradeable response in the placebo is a known caveat.
        \end{itemize}

    \fontfamily{cmr}
    \frametitle{What This Design Replaces}
        \footnotesize
        The unified model subsumes three earlier framings into a single regression:
        \medskip
        \begin{itemize}\setlength{\itemsep}{0.05cm}
            \item \textbf{Block A / B / C separate sections}: collapsed
            into outcome rows of one regression.
            Block A's ``log(Resident workers)'' is now the Inc-bind column on
            \texttt{log\_live\_and\_work}; Block C's
            ``log(Residents employed)'' is the Inc-bind column on
            \texttt{log\_total\_residents}.
            \item \textbf{Piecewise pair-flow model}: cousub-pair regression
            with binding-state transitions. Saturated by pair FE; aggregate null.
            Replaced by the binding decomposition at the muni level, where
            sample sizes are tractable and signal survives.
            \item \textbf{Q3 asymmetry section}: the Inc/Dec direction split
            is now a within-design dimension of the cell taxonomy.
        \end{itemize}
    \end{frame}

    \begin{frame}
    \fontfamily{cmr}
    \frametitle{Summary}
        \begin{enumerate}\setlength{\itemsep}{0.08cm}\footnotesize
            \item \textbf{Causal effect on residential sorting: yes.}
            Muni Inc-bind cell, every estimator: consistent negative on
            log(live\_and\_work) and log(total\_residents); positive on
            share commuting out.
            \item \textbf{Within-design placebo passes.} Muni Inc-nobind
            cell --- rate hike that doesn't move the bill --- null behavior
            on every level outcome. The institutional rule, not generic
            confounding, identifies the response.
            \item \textbf{Three rates, three populations.} Muni rate
            moves residents (live-and-work, total residents); SD rate
            moves no one; nonres rate moves in-commuters.
            SD doesn't sort \emph{even within the binding decomposition} ---
            stronger version of the prior block-level-LODES finding.
            \item \textbf{Magnitude (stacked continuous, headline).}
            1pp \emph{muni}-rate hike when home binds $\Rightarrow$
            $-39\%$ live-and-work, $-21\%$ in-commuters, $-26\%$ total jobs,
            $-10\%$ total residents, $+4.1$pp share commuting out.
            \item \textbf{Robust to treatment specification.} Muni-only,
            home-stack combined, and muni+SD-separate all give the same
            order of magnitude on Inc-bind / log(live\_and\_work).
            \item \textbf{Industry decomposition (issue \#2).} Within Inc-bind:
            consumer-facing jobs $-29\%^{**}$; tradeable jobs $+0.244$ (insig).
            Composition shifts toward tradeables. Direct evidence for
            local-services-demand spillover.
            \item \textbf{Spillover (issue \#3).} Ring-1 of Inc-bind munis
            loses live\_and\_work ($-15\%$); ring-2 gains ($+9\%$). Pattern
            unchanged under lower-rate restriction --- consistent with
            spatial labor-market clustering, not pure fiscal-advantage
            migration.
        \end{enumerate}
    \end{frame}
